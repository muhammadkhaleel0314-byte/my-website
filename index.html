packages = ["pandas", "openpyxl", "fpdf2", "plotly"] import io import pandas as pd from fpdf import FPDF import js from pyodide.ffi import create_proxy global_df = None pivot_df = None def handle_file_upload(event): global global_df, pivot_df file_list = event.target.files if file_list.length == 0: return file = file_list.item(0) async def process(): global global_df, pivot_df array_buffer = await file.arrayBuffer() bytes_data = bytes(array_buffer.to_py()) # Read Excel df = pd.read_excel(io.BytesIO(bytes_data)) df['recovery_date'] = pd.to_datetime(df['recovery_date'], errors='coerce') df.dropna(subset=['recovery_date'], inplace=True) df['day'] = df['recovery_date'].dt.day def get_range(day): if 1 <= day <= 10: return "1-10" elif 11 <= day <= 20: return "11-20" elif 21 <= day <= 31: return "21-31" return "Unknown" df['range'] = df['day'].apply(get_range) global_df = df # Populate Branch Dropdown branches = sorted(df['branch_id'].dropna().unique().tolist()) select_el = js.document.getElementById("branch-select") select_el.innerHTML = '' for b in branches: opt = js.document.createElement("option") opt.value = str(b) opt.text = f"Branch {b}" select_el.appendChild(opt) js.document.getElementById("filter-container").classList.remove("hidden") js.document.getElementById("download-container").classList.remove("hidden") # Pivot Table pivot_df = df.groupby(['branch_id', 'project', 'recovery_date']).agg( Receipts=('receipt_no', 'count'), Amount=('amount', 'sum') ).reset_index() # Render Data Preview on Page output_div = js.document.getElementById("output-area") output_div.innerHTML = f"""

📄 Complete Recovery Data (Total Rows: {len(df)})

{''.join([f"" for _, r in df.head(10).iterrows()])}
Branch Project Date Amount
{r.branch_id}{r.project}{str(r.recovery_date)[:10]}Rs {r.amount:,.0f}
""" js.window.eval("async function run() {}") import asyncio asyncio.ensure_future(process()) # Attach Event Listener to File Input file_input = js.document.getElementById("excel-file") file_input.addEventListener("change", create_proxy(handle_file_upload)) class PDF(FPDF): def header(self): pass def footer(self): pass def generate_pdf(event): global pivot_df if pivot_df is None: return selected_branch = js.document.getElementById("branch-select").value pdf = PDF() pdf.set_auto_page_break(auto=True, margin=15) if selected_branch == "ALL": target_branches = pivot_df['branch_id'].unique() else: # Handle type matching if branch_id is numeric/string try: target_branches = [type(pivot_df['branch_id'].iloc[0])(selected_branch)] except: target_branches = [selected_branch] for branch in target_branches: branch_df = pivot_df[pivot_df['branch_id'] == branch] if branch_df.empty: continue pdf.add_page() pdf.set_font("Arial", 'B', 14) pdf.cell(0, 10, f"Branch: {branch}", ln=True, align='C') branch_total_amount = 0 branch_total_receipts = 0 for project, proj_df in branch_df.groupby('project'): pdf.set_font("Arial", 'B', 12) pdf.cell(0, 8, f"Project: {project}", ln=True) pdf.set_font("Arial", 'B', 10) pdf.cell(40, 8, "Date", border=1, align='C') pdf.cell(40, 8, "Receipts", border=1, align='C') pdf.cell(40, 8, "Amount", border=1, align='C') pdf.ln() project_total_amount = 0 project_total_receipts = 0 pdf.set_font("Arial", '', 10) for _, row in proj_df.iterrows(): date_str = row['recovery_date'].strftime('%Y-%m-%d') if pd.notnull(row['recovery_date']) else '' pdf.cell(40, 8, date_str, border=1) pdf.cell(40, 8, str(row['Receipts']), border=1, align='C') pdf.cell(40, 8, f"Rs {row['Amount']:,.0f}", border=1, align='R') pdf.ln() project_total_receipts += row['Receipts'] project_total_amount += row['Amount'] pdf.set_font("Arial", 'B', 10) pdf.cell(40, 8, "Project Total", border=1) pdf.cell(40, 8, str(project_total_receipts), border=1, align='C') pdf.cell(40, 8, f"Rs {project_total_amount:,.0f}", border=1, align='R') pdf.ln(10) branch_total_receipts += project_total_receipts branch_total_amount += project_total_amount pdf.set_font("Arial", 'B', 11) pdf.cell(40, 8, "Branch Total", border=1) pdf.cell(40, 8, str(branch_total_receipts), border=1, align='C') pdf.cell(40, 8, f"Rs {branch_total_amount:,.0f}", border=1, align='R') # Output PDF bytes pdf_bytes = pdf.output(dest='S').encode('latin1') # Trigger Browser Download via JS import base64 b64_encoded = base64.b64encode(pdf_bytes).decode('utf-8') filename = f"Branch_{selected_branch}.pdf" js.eval(f""" const link = document.createElement('a'); link.href = 'data:application/pdf;base64,{b64_encoded}'; link.download = '{filename}'; document.body.appendChild(link); link.click(); document.body.removeChild(link); """) download_btn = js.document.getElementById("download-btn") download_btn.addEventListener("click", create_proxy(generate_pdf))
